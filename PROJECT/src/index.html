/* LOGIN */ function login() { let username = document.getElementById("username").value; let password = document.getElementById("password").value; if (username === "" || password === "") { document.getElementById("loginMessage").textContent = "Please enter username and password."; return; } window.location.href = "patient.html"; } /* PATIENT MONITORING */ function monitorPatient() { let patientId = document.getElementById("patientId").value; let heartRate = Number(document.getElementById("heartRate").value); let bloodPressure = document.getElementById("bloodPressure").value; let temperature = Number(document.getElementById("temperature").value); let oxygenLevel = Number(document.getElementById("oxygenLevel").value); /* Check input */ if ( !patientId || !heartRate || !bloodPressure || !temperature || !oxygenLevel ) { alert("Please enter all patient details."); return; } /* Read BP */ let bp = bloodPressure.split("/"); if ( bp.length !== 2 || isNaN(Number(bp[0])) || isNaN(Number(bp[1])) ) { alert("Please enter BP like 120/80."); return; } let systolicBP = Number(bp[0]); let diastolicBP = Number(bp[1]); let status; let condition; let action; /* EMERGENCY */ if ( heartRate > 120 || systolicBP > 140 || diastolicBP > 90 || temperature > 39 || oxygenLevel < 90 ) { status = "EMERGENCY"; condition = "Abnormal vital signs detected"; action = "Immediate medical attention and doctor notification required."; } /* WARNING */ else if ( heartRate < 60 || heartRate > 100 || systolicBP > 130 || diastolicBP > 80 || temperature > 37 || oxygenLevel < 95 ) { status = "WARNING"; condition = "One or more vital signs are outside the normal range"; action = "Keep patient under observation and continue monitoring."; } /* NORMAL */ else { status = "NORMAL"; condition = "Vital signs are within the normal range"; action = "Continue regular monitoring."; } /* Save data */ let patientData = { patientId: patientId, heartRate: heartRate, systolicBP: systolicBP, diastolicBP: diastolicBP, temperature: temperature, oxygenLevel: oxygenLevel, status: status, condition: condition, action: action, time: new Date().toLocaleString() }; localStorage.setItem( "patientData", JSON.stringify(patientData) ); /* Go to result page */ window.location.href = "result.html"; } /* RESULT PAGE */ function loadResult() { let data = JSON.parse( localStorage.getItem("patientData") ); if (!data) { return; } document.getElementById("resultId").textContent = data.patientId; document.getElementById("resultHeartRate").textContent = data.heartRate + " BPM"; document.getElementById("resultBP").textContent = data.systolicBP + "/" + data.diastolicBP + " mmHg"; document.getElementById("resultTemp").textContent = data.temperature + " °C"; document.getElementById("resultOxygen").textContent = data.oxygenLevel + "%"; document.getElementById("condition").textContent = data.condition; document.getElementById("action").textContent = data.action; document.getElementById("monitoringTime").textContent = data.time; document.getElementById("status").textContent = data.status; let statusCard = document.getElementById("statusCard"); if (data.status === "NORMAL") { statusCard.classList.add( "status-normal" ); } else if (data.status === "WARNING") { statusCard.classList.add( "status-warning" ); } else { statusCard.classList.add( "status-emergency" ); } } /* DOCTOR ALERT */ function loadDoctorAlert() { let data = JSON.parse( localStorage.getItem("patientData") ); if (!data) { return; } document.getElementById("doctorStatus").textContent = data.status; document.getElementById("doctorId").textContent = data.patientId; document.getElementById("doctorHeartRate").textContent = data.heartRate + " BPM"; document.getElementById("doctorBP").textContent = data.systolicBP + "/" + data.diastolicBP + " mmHg"; document.getElementById("doctorTemp").textContent = data.temperature + " °C"; document.getElementById("doctorOxygen").textContent = data.oxygenLevel + "%"; document.getElementById("doctorCondition").textContent = data.condition; document.getElementById("doctorAction").textContent = data.action; } /* GO TO DOCTOR ALERT */ function goToDoctorAlert() { window.location.href = "doctor.html"; } /* GO BACK TO PATIENT PAGE */ function goToPatientPage() { window.location.href = "patient.html"; } /* Automatically load required page data */ if (window.location.pathname.includes("result.html")) { loadResult(); } if (window.location.pathname.includes("doctor.html")) { loadDoctorAlert(); }
